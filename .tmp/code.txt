/** להריץ פעם אחת מתוך עורך הסקריפט. */
function setup() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  if (!ss) throw new Error('יש לפתוח את הסקריפט מתוך הגיליון: הרחבות ← Apps Script.');
  ss.setSpreadsheetTimeZone(CONFIG.TIMEZONE);
  const props = PropertiesService.getScriptProperties();
  props.setProperty('SPREADSHEET_ID', ss.getId());

  // גיליון מפרסמים
  let sh = ss.getSheetByName(SHEET_NAME);
  if (!sh) {
    sh = ss.getSheets()[0];
    if (sh.getLastRow() > 0) sh = ss.insertSheet(SHEET_NAME); else sh.setName(SHEET_NAME);
  }
  sh.setRightToLeft(true);
  sh.getRange(1, 1, 1, COLS.length).setValues([COLS])
    .setFontWeight('bold').setBackground('#1f2a44').setFontColor('#ffffff');
  sh.setFrozenRows(1);
  sh.setFrozenColumns(C['שם העסק']);

  const statusRule = SpreadsheetApp.newDataValidation()
    .requireValueInList(Object.values(STATUS), true).setAllowInvalid(false).build();
  sh.getRange(2, C['סטטוס'], 1000, 1).setDataValidation(statusRule);

  const dateRule = SpreadsheetApp.newDataValidation().requireDate().setAllowInvalid(false).build();
  [C['תחילת פרסום'], C['ריענון אחרון']].forEach(col =>
    sh.getRange(2, col, 1000, 1).setDataValidation(dateRule).setNumberFormat('dd/mm/yyyy'));
  sh.getRange(2, C['תאריך הרשמה'], 1000, 1).setNumberFormat('dd/mm/yyyy hh:mm');
  sh.getRange(2, C['ריענון הבא'], 1000, 1).setNumberFormat('dd/mm/yyyy');

  // צבעי סטטוס
  const statusRange = sh.getRange(2, C['סטטוס'], 1000, 1);
  const colors = [
    [STATUS.PENDING, '#fff1c2'], [STATUS.ACTIVE, '#cdeccf'],
    [STATUS.PAUSED, '#e3e3e3'], [STATUS.ENDED, '#f4c7c3'], [STATUS.IRRELEVANT, '#eeeeee'],
  ];
  sh.setConditionalFormatRules(colors.map(([text, bg]) =>
    SpreadsheetApp.newConditionalFormatRule().whenTextEqualTo(text)
      .setBackground(bg).setRanges([statusRange]).build()));

  sh.setColumnWidths(1, COLS.length, 130);
  sh.setColumnWidth(C['רעיון / טקסט למודעה'], 260);
  sh.setColumnWidth(C['הערות פנימיות'], 220);

  buildDashboard_(ss);
  getRootFolder_();

  // טריגר יומי (מוחק כפילויות קודמות)
  ScriptApp.getProjectTriggers()
    .filter(t => t.getHandlerFunction() === 'dailyReminders')
    .forEach(t => ScriptApp.deleteTrigger(t));
  ScriptApp.newTrigger('dailyReminders').timeBased().everyDays(1)
    .atHour(CONFIG.REMINDER_HOUR).inTimezone(CONFIG.TIMEZONE).create();

  console.log('ההתקנה הושלמה ✔');
  try { SpreadsheetApp.getUi().alert('ההתקנה הושלמה ✔\n\nהשלב הבא: פריסה ← פריסה חדשה ← אפליקציית אינטרנט.'); } catch (_) {}
}

function buildDashboard_(ss) {
  let d = ss.getSheetByName(DASH_NAME) || ss.insertSheet(DASH_NAME);
  d.clear();
  d.setRightToLeft(true);
  const s = `'${SHEET_NAME}'!`;
  const status = `${s}${colLetter_(C['סטטוס'])}2:${colLetter_(C['סטטוס'])}`;
  const nextRef = `${s}${colLetter_(C['ריענון הבא'])}2:${colLetter_(C['ריענון הבא'])}`;
  const priceRef = `${s}${colLetter_(C['מחיר חודשי'])}2:${colLetter_(C['מחיר חודשי'])}`;
  const rows = [
    ['לוח בקרה — ' + CONFIG.BRAND, ''],
    ['', ''],
    ['מפרסמים פעילים', `=COUNTIF(${status},"${STATUS.ACTIVE}")`],
    ['ממתינים לתשלום', `=COUNTIF(${status},"${STATUS.PENDING}")`],
    ['מושהים', `=COUNTIF(${status},"${STATUS.PAUSED}")`],
    ['משבצות פנויות', `=${CONFIG.MAX_SLOTS}-B3`],
    ['הכנסה חודשית (לפני מע״מ)', `=SUMIFS(${priceRef},${status},"${STATUS.ACTIVE}")+COUNTIFS(${status},"${STATUS.ACTIVE}",${priceRef},"")*${CONFIG.MONTHLY_PRICE}`],
    ['ריענונים ב־7 הימים הקרובים', `=COUNTIFS(${status},"${STATUS.ACTIVE}",${nextRef},"<="&TODAY()+7)`],
    ['הופעות בשעה לכל מפרסם (בערך)', `=IF(B3=0,"-",ROUND(3600/(MAX(B3,1)*12)))`],
  ];
  d.getRange(1, 1, rows.length, 2).setValues(rows);
  d.getRange('A1').setFontSize(16).setFontWeight('bold');
  d.getRange('A3:A9').setFontWeight('bold');
  d.getRange('B7').setNumberFormat('₪#,##0');
  d.getRange('B3:B9').setFontSize(14).setHorizontalAlignment('center');
  d.setColumnWidth(1, 260); d.setColumnWidth(2, 140);
}

function doGet(e) {
  if (e && e.parameter && e.parameter.health !== undefined) return json_({ ok: true, service: CONFIG.BRAND, message: 'המערכת פעילה' });
  return adminPage_();
}

/** כל בקשת POST: או קריאה מלוח הניהול (rpc), או הרשמה חדשה מהטופס */
function doPost(e) {
  let body = {};
  try { body = JSON.parse((e && e.postData && e.postData.contents) || '{}'); } catch (_) {}
  if (body && body.rpc) return json_(rpc_(body.rpc, body.args));
  return handleSignup_(e);
}

// הפונקציות היחידות שלוח הניהול יכול להפעיל מבחוץ. כולן (חוץ מהכניסה) דורשות טוקן תקף.
const RPC_FUNCS = {
  adminAuthState: () => adminAuthState(),
  adminLogin: a => adminLogin(a[0]),
  adminSetPassword: a => adminSetPassword(a[0]),
  adminLogout: a => adminLogout(a[0]),
  getAdminData: a => getAdminData(a[0]),
  adminSetStatus: a => adminSetStatus(a[0], a[1], a[2]),
  adminMarkRefresh: a => adminMarkRefresh(a[0], a[1]),
  adminMarkPaid: a => adminMarkPaid(a[0], a[1]),
  adminAddDays: a => adminAddDays(a[0], a[1], a[2]),
  adminSetDeal: a => adminSetDeal(a[0], a[1], a[2], a[3], a[4], a[5]),
  adminSaveNote: a => adminSaveNote(a[0], a[1], a[2]),
  adminThumb: a => adminThumb(a[0], a[1]),
  adminRecordPayment: a => adminRecordPayment(a[0], a[1], a[2], a[3], a[4], a[5]),
  adminIssueReceipt: a => adminIssueReceipt(a[0], a[1]),
};

function rpc_(name, args) {
  const fn = Object.prototype.hasOwnProperty.call(RPC_FUNCS, name) ? RPC_FUNCS[name] : null;
  if (!fn) return { ok: false, error: 'פעולה לא מוכרת' };
  try {
    return { ok: true, result: fn(Array.isArray(args) ? args : []) };
  } catch (err) {
    const msg = String(err && err.message || err);
    if (msg !== 'AUTH') console.error(name, msg);
    return { ok: false, error: msg };
  }
}

function handleSignup_(e) {
  const lock = LockService.getScriptLock();
  try {
    lock.waitLock(20000);
    const body = JSON.parse((e && e.postData && e.postData.contents) || '{}');

    // מלכודת ספאם: שדה מוסתר שבני אדם לא ממלאים
    if (body.website) return json_({ ok: true, id: 'X' });

    const f = {
      business: clean_(body.business, 120),
      businessId: clean_(body.businessId, 20),
      category: clean_(body.category, 80),
      address: clean_(body.address, 200),
      contact: clean_(body.contact, 80),
      phone: clean_(body.phone, 20),
      email: clean_(body.email, 120),
      adText: clean_(body.adText, 1500),
      notes: clean_(body.notes, 1000),
      terms: body.terms === true,
      pay: Object.values(PAY).indexOf(body.pay) >= 0 ? body.pay : PAY.CARD,
    };
    const missing = ['business', 'contact', 'phone'].filter(k => !f[k]);
    if (missing.length) return json_({ ok: false, error: 'חסרים שדות חובה' });
    if (!f.terms) return json_({ ok: false, error: 'יש לאשר את תנאי ההתקשרות' });
    if (!/^0\d{8,9}$/.test(f.phone.replace(/\D/g, ''))) return json_({ ok: false, error: 'מספר טלפון לא תקין' });

    const files = Array.isArray(body.files) ? body.files.slice(0, 6) : [];
    let totalBytes = 0;
    const blobs = files.map(file => {
      const type = String(file.type || '');
      if (!/^(image\/|video\/|application\/pdf)/.test(type)) throw new Error('סוג קובץ לא נתמך');
      const bytes = Utilities.base64Decode(String(file.data || ''));
      totalBytes += bytes.length;
      const name = String(file.name || 'file').replace(/[\\/:*?"<>|]/g, '_').slice(0, 100);
      return Utilities.newBlob(bytes, type, name);
    });
    if (totalBytes > CONFIG.MAX_UPLOAD_MB * 1024 * 1024) return json_({ ok: false, error: 'הקבצים גדולים מדי' });

    const sh = getSheet_();
    const id = nextId_(sh);
    const now = new Date();

    let folderUrl = '';
    if (blobs.length) {
      const folder = getRootFolder_().createFolder(`${id} — ${f.business}`);
      blobs.forEach(b => folder.createFile(b));
      folderUrl = folder.getUrl();
    }

    const row = new Array(COLS.length).fill('');
    row[C['מזהה'] - 1] = id;
    row[C['תאריך הרשמה'] - 1] = now;
    row[C['סטטוס'] - 1] = STATUS.PENDING;
    row[C['שם העסק'] - 1] = safe_(f.business);
    row[C['ח״פ / עוסק'] - 1] = safe_(f.businessId);
    row[C['תחום'] - 1] = safe_(f.category);
    row[C['כתובת העסק'] - 1] = safe_(f.address);
    row[C['איש קשר'] - 1] = safe_(f.contact);
    row[C['טלפון'] - 1] = "'" + f.phone;
    row[C['מייל'] - 1] = safe_(f.email);
    row[C['רעיון / טקסט למודעה'] - 1] = safe_(f.adText);
    row[C['הערות הלקוח'] - 1] = safe_(f.notes);
    row[C['קבצים'] - 1] = folderUrl ? `=HYPERLINK("${folderUrl}","תיקייה (${blobs.length})")` : 'לא הועלו';
    row[C['אישר תנאים'] - 1] = Utilities.formatDate(now, CONFIG.TIMEZONE, 'dd/MM/yyyy HH:mm');
    row[C['אמצעי תשלום'] - 1] = f.pay;
    sh.appendRow(row);

    const r = sh.getLastRow();
    const S = colLetter_(C['סטטוס']), N = colLetter_(C['תחילת פרסום']), L = colLetter_(C['ריענון אחרון']);
    sh.getRange(r, C['ריענון הבא']).setFormula(
      `=IF(${S}${r}<>"${STATUS.ACTIVE}","",IF(${L}${r}<>"",${L}${r}+${CONFIG.REFRESH_DAYS},IF(${N}${r}<>"",${N}${r}+${CONFIG.REFRESH_DAYS},"")))`);

    notifyNewSignup_(id, f, blobs.length, folderUrl);
    return json_({ ok: true, id });
  } catch (err) {
    console.error(err);
    return json_({ ok: false, error: 'אירעה שגיאה בשמירה. אפשר לנסות שוב או לפנות בוואטסאפ.' });
  } finally {
    try { lock.releaseLock(); } catch (_) {}
  }
}

function dailyReminders() { runReminders_(false); }

/** מהתפריט בגיליון: שולח את המייל גם אם אין משימות, וגם בשבת. */
function sendRemindersNow() {
  runReminders_(true);
  SpreadsheetApp.getUi().alert('מייל המשימות נשלח אל ' + CONFIG.OWNER_EMAIL);
}

function runReminders_(manual) {
  const dow = Number(Utilities.formatDate(new Date(), CONFIG.TIMEZONE, 'u')); // 6 = שבת
  if (!manual && dow === 6) return;

  const sh = getSheet_();
  const last = sh.getLastRow();
  if (last < 2) { if (manual) sendTasks_([], [], [], []); return; }
  const data = sh.getRange(2, 1, last - 1, COLS.length).getValues();
  const today = startOfDay_(new Date());
  const now = new Date();

  const unpaid = [], refresh = [], noStart = [], payDue = [];
  data.forEach(row => {
    const g = name => row[C[name] - 1];
    const status = g('סטטוס');
    const item = {
      id: g('מזהה'), business: g('שם העסק'), contact: g('איש קשר'),
      phone: String(g('טלפון')).replace(/^'/, ''),
    };
    if (!item.business) return;

    if (status === STATUS.PENDING) {
      const reg = g('תאריך הרשמה');
      if (reg instanceof Date && (now - reg) / 36e5 >= CONFIG.PAYMENT_REMIND_AFTER_HOURS) {
        item.days = Math.floor((today - startOfDay_(reg)) / 864e5);
        unpaid.push(item);
      }
    }
    if (status === STATUS.ACTIVE && g('אמצעי תשלום') && g('אמצעי תשלום') !== PAY.CARD) {
      const pu = g('שולם עד');
      const left = pu instanceof Date ? Math.round((startOfDay_(pu) - today) / 864e5) : null;
      if (left === null || left <= CONFIG.PAY_DUE_WARN_DAYS) {
        payDue.push(Object.assign({}, item, { left, method: g('אמצעי תשלום'), price: Number(g('מחיר חודשי')) || CONFIG.MONTHLY_PRICE }));
      }
    }
    if (status === STATUS.ACTIVE) {
      const start = g('תחילת פרסום'), lastRef = g('ריענון אחרון');
      const base = lastRef instanceof Date ? lastRef : (start instanceof Date ? start : null);
      if (!base) { noStart.push(item); return; }
      const due = new Date(startOfDay_(base).getTime() + CONFIG.REFRESH_DAYS * 864e5);
      const daysLeft = Math.round((due - today) / 864e5);
      if (daysLeft <= CONFIG.REFRESH_WARN_DAYS) { item.daysLeft = daysLeft; refresh.push(item); }
    }
  });

  if (!manual && !unpaid.length && !refresh.length && !noStart.length && !payDue.length) return;
  sendTasks_(unpaid, refresh, noStart, payDue);
}

function sendTasks_(unpaid, refresh, noStart, payDue) {
  payDue = payDue || [];
  const ssUrl = SpreadsheetApp.openById(getSsId_()).getUrl();
  const pay = CONFIG.PAYMENT_URL ? `\nקישור לתשלום: ${CONFIG.PAYMENT_URL}` : '';
  const section = (title, items, render) => !items.length ? '' :
    `<h3 style="margin:24px 0 8px">${title} (${items.length})</h3>` + items.map(render).join('');

  const body =
    section('🏦 תשלום חודשי בהעברה / ביט', payDue, it => card_(it,
      it.left === null ? 'אין תאריך "שולם עד"' : (it.left < 0 ? `באיחור של ${-it.left} ימים` : (it.left === 0 ? 'היום' : `בעוד ${it.left} ימים`)),
      payDueMsg_(it),
      'אחרי שהכסף נכנס: בלוח הניהול ← "התקבל תשלום".')) +

const ADMIN_SESSION_DAYS = 30;

function hashPw_(pw, salt) {
  const bytes = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, salt + '|' + pw, Utilities.Charset.UTF_8);
  return Utilities.base64Encode(bytes);
}
function sessions_() {
  try { return JSON.parse(PropertiesService.getScriptProperties().getProperty('ADMIN_SESSIONS') || '{}'); } catch (_) { return {}; }
}
function saveSessions_(s) {
  const now = Date.now();
  const keep = Object.keys(s).filter(k => s[k] > now).sort((a, b) => s[b] - s[a]).slice(0, 10);
  const out = {}; keep.forEach(k => { out[k] = s[k]; });
  PropertiesService.getScriptProperties().setProperty('ADMIN_SESSIONS', JSON.stringify(out));
}
function assertOwner_(token) {
  if (token && typeof token === 'string') {
    const exp = sessions_()[token];
    if (exp && exp > Date.now()) return;
  }
  throw new Error('AUTH');
}

/** להריץ מהעורך כדי לפתוח 30 דקות להגדרת סיסמה (או החלפתה) מתוך הלוח. */
function openAdminSetup() {
  PropertiesService.getScriptProperties().setProperty('ADMIN_SETUP_UNTIL', String(Date.now() + 30 * 60 * 1000));
  console.log('אפשר להגדיר סיסמה בלוח הניהול ב־30 הדקות הקרובות.');
}

function adminAuthState() {
  const p = PropertiesService.getScriptProperties();
  return { hasPassword: !!p.getProperty('ADMIN_HASH'), setupOpen: Number(p.getProperty('ADMIN_SETUP_UNTIL') || 0) > Date.now() };
}

function newSession_() {
  const token = Utilities.getUuid() + Utilities.getUuid().slice(0, 8);
  const s = sessions_(); s[token] = Date.now() + ADMIN_SESSION_DAYS * 864e5; saveSessions_(s);
  return { token };
}

function adminSetPassword(pw) {
  const lock = LockService.getScriptLock(); lock.waitLock(10000);
  try {
    if (!adminAuthState().setupOpen) throw new Error('הגדרת הסיסמה סגורה. יש להריץ את openAdminSetup בעורך.');
    pw = String(pw || '');
    if (pw.length < 8) throw new Error('הסיסמה צריכה להיות לפחות 8 תווים.');
    const p = PropertiesService.getScriptProperties();
    const salt = Utilities.getUuid();
    p.setProperty('ADMIN_SALT', salt);
    p.setProperty('ADMIN_HASH', hashPw_(pw, salt));
    p.deleteProperty('ADMIN_SETUP_UNTIL');
    p.deleteProperty('ADMIN_SESSIONS');
    return newSession_();
  } finally { lock.releaseLock(); }
}

function adminLogin(pw) {
  const cache = CacheService.getScriptCache();
  const fails = Number(cache.get('admin_fails') || 0);
  if (fails >= 8) throw new Error('יותר מדי ניסיונות. אפשר לנסות שוב בעוד 15 דקות.');
  const p = PropertiesService.getScriptProperties();
  const hash = p.getProperty('ADMIN_HASH');
  if (!hash) throw new Error('עוד לא הוגדרה סיסמה.');
  if (hashPw_(String(pw || ''), p.getProperty('ADMIN_SALT')) !== hash) {
    cache.put('admin_fails', String(fails + 1), 900);
    Utilities.sleep(800);
    throw new Error('סיסמה שגויה.');
  }
  cache.remove('admin_fails');
  return newSession_();
}

function adminLogout(token) {
  const s = sessions_(); delete s[token]; saveSessions_(s);
  return true;
}

function getAdminData(token) {
  assertOwner_(token);
  const ss = SpreadsheetApp.openById(getSsId_());
  const sh = ensureCols_(ss.getSheetByName(SHEET_NAME));
  const last = sh.getLastRow();
  const today = startOfDay_(new Date());
  const now = new Date();
  const items = [];
  const sbOn = sbConfigured_();

  if (last >= 2) {
    const rng = sh.getRange(2, 1, last - 1, COLS.length);
    const vals = rng.getValues();
    const forms = rng.getFormulas();
    vals.forEach((row, i) => {
      const g = n => row[C[n] - 1];
      if (!g('שם העסק')) return;
      const m = String(forms[i][C['קבצים'] - 1] || '').match(/folders\/([\w-]+)/);
      const it = {
        id: String(g('מזהה')), status: String(g('סטטוס')), business: String(g('שם העסק')),
        businessId: String(g('ח״פ / עוסק')), category: String(g('תחום')), address: String(g('כתובת העסק')),
        contact: String(g('איש קשר')), phone: String(g('טלפון')).replace(/^'/, ''), email: String(g('מייל')),
        adText: String(g('רעיון / טקסט למודעה')), notes: String(g('הערות הלקוח')), internal: String(g('הערות פנימיות')),
        folderId: m ? m[1] : '', folderUrl: m ? 'https://drive.google.com/drive/folders/' + m[1] : '',
        registered: iso_(g('תאריך הרשמה')), start: iso_(g('תחילת פרסום')), lastRefresh: iso_(g('ריענון אחרון')),
        nextRefresh: '', daysLeft: null, tasks: [],
        price: Number(g('מחיר חודשי')) || CONFIG.MONTHLY_PRICE, customPrice: Number(g('מחיר חודשי')) > 0,
        commitEnd: iso_(g('התחייבות עד')),
        pay: String(g('אמצעי תשלום') || PAY.CARD), paidUntil: iso_(g('שולם עד')),
        lastPay: (() => { try { return JSON.parse(g('תשלום אחרון') || 'null'); } catch (_) { return null; } })(),
        receiptNo: String(g('קבלה אחרונה') || ''), receiptUrl: String(g('קישור קבלה') || ''),
      };
      it.receiptMissing = !!(it.lastPay && it.lastPay.method !== PAY.CARD && !it.receiptUrl);
      it.priceVat = withVat_(it.price);
      if (it.status === STATUS.ACTIVE && it.pay !== PAY.CARD) {
        const pu = g('שולם עד');
        const left = pu instanceof Date ? Math.round((startOfDay_(pu) - today) / 864e5) : null;
        if (left === null || left <= CONFIG.PAY_DUE_WARN_DAYS) {
          const when = left === null ? 'חסר תאריך "שולם עד"' : (left < 0 ? 'באיחור של ' + (-left) + ' ימים' : (left === 0 ? 'היום' : 'בעוד ' + left + ' ימים'));
          it.tasks.push({ type: 'paydue', label: 'תשלום ב' + it.pay + ' ' + when, wa: waLink_(it.phone, payDueMsg_({ contact: it.contact, business: it.business, price: it.price, method: it.pay })) });
        }
      }
      const ce = g('התחייבות עד');
      if (it.status === STATUS.ACTIVE && ce instanceof Date) {
        const left = Math.round((startOfDay_(ce) - today) / 864e5);
        if (left <= 14) {
          const when = left < 0 ? 'הסתיימה לפני ' + (-left) + ' ימים' : (left === 0 ? 'מסתיימת היום' : 'מסתיימת בעוד ' + left + ' ימים');
          it.tasks.push({ type: 'commit', label: 'התחייבות ' + when + ' · לחדש או להחזיר למחיר רגיל', wa: waLink_(it.phone, adminMsg_('commit', it)) });
        }
      }
      if (it.status === STATUS.PENDING) {
        const reg = g('תאריך הרשמה');
        if (reg instanceof Date && (now - reg) / 36e5 >= CONFIG.PAYMENT_REMIND_AFTER_HOURS) {
          const days = Math.floor((today - startOfDay_(reg)) / 864e5);
          it.tasks.push({ type: 'unpaid', label: 'נרשם לפני ' + days + ' ימים ועוד לא שילם', wa: waLink_(it.phone, adminMsg_('unpaid', it)) });
        }
      }
      if (it.status === STATUS.ACTIVE) {
        const s = g('תחילת פרסום'), l = g('ריענון אחרון');
        const base = l instanceof Date ? l : (s instanceof Date ? s : null);
        if (!base) {
          it.tasks.push({ type: 'nostart', label: 'חסר תאריך תחילת פרסום', wa: waLink_(it.phone, adminMsg_('nostart', it)) });
        } else {
          const due = new Date(startOfDay_(base).getTime() + CONFIG.REFRESH_DAYS * 864e5);
          it.nextRefresh = iso_(due);
          it.daysLeft = Math.round((due - today) / 864e5);
          if (it.daysLeft <= CONFIG.REFRESH_WARN_DAYS) {
            const when = it.daysLeft < 0 ? 'באיחור של ' + (-it.daysLeft) + ' ימים' : (it.daysLeft === 0 ? 'היום' : 'בעוד ' + it.daysLeft + ' ימים');
            it.tasks.push({ type: 'refresh', label: 'ריענון חודשי ' + when, wa: waLink_(it.phone, adminMsg_('refresh', it)) });
          }
        }
      }
      if (sbOn && it.receiptMissing) it.tasks.push({ type: 'receipt', label: 'חסרה חשבונית לתשלום האחרון (' + it.lastPay.date + ')' });
      it.waChat = waLink_(it.phone, adminMsg_('chat', it));
      items.push(it);
    });
  }

  items.reverse();
  const count = s => items.filter(x => x.status === s).length;
  const active = count(STATUS.ACTIVE);
  return {
    items,
    stats: {
      active, pending: count(STATUS.PENDING), paused: count(STATUS.PAUSED),
      free: Math.max(0, CONFIG.MAX_SLOTS - active), maxSlots: CONFIG.MAX_SLOTS,
      revenue: items.filter(x => x.status === STATUS.ACTIVE).reduce((a, x) => a + x.price, 0),
      defaultPrice: CONFIG.MONTHLY_PRICE,
      perHour: active ? Math.round(3600 / (active * 12)) : null,
      tasks: items.reduce((a, x) => a + x.tasks.length, 0),
    },
    statuses: Object.values(STATUS),
    payMethods: Object.values(PAY),
    sb: { on: sbConfigured_(), env: sbEnv_() },
    vat: CONFIG.VAT,
    sheetUrl: ss.getUrl(),
    today: iso_(new Date()),
  };
}

function findRowById_(sh, id) {
  const last = sh.getLastRow();
  if (last < 2) return 0;
  const ids = sh.getRange(2, C['מזהה'], last - 1, 1).getValues();
  for (let i = 0; i < ids.length; i++) if (String(ids[i][0]) === String(id)) return i + 2;
  return 0;
}

  const sh = getSheet_();
  const r = findRowById_(sh, id);
  if (!r) throw new Error('המפרסם לא נמצא');
  sh.getRange(r, C['סטטוס']).setValue(status);
  if (status === STATUS.ACTIVE) {
    const start = sh.getRange(r, C['תחילת פרסום']);
    if (!start.getValue()) start.setValue(startOfDay_(new Date()));
    const method = sh.getRange(r, C['אמצעי תשלום']).getValue();

    if (method && method !== PAY.CARD && !(paid.getValue() instanceof Date)) paid.setValue(addMonths_(start.getValue(), 1));
  }
  return getAdminData(token);
}

function adminSetDeal(token, id, price, commitEnd, pay, paidUntil) {
  assertOwner_(token);
  const sh = getSheet_();
  const r = findRowById_(sh, id);
  if (!r) throw new Error('המפרסם לא נמצא');

  sh.getRange(r, C['מחיר חודשי']).setValue(p === '' || p === CONFIG.MONTHLY_PRICE ? '' : Math.round(p));

function adminThumb(token, folderId) {
  assertOwner_(token);
  if (!folderId) return '';
  const cache = CacheService.getScriptCache();
  const hit = cache.get('th_' + folderId);
  if (hit) return hit === '-' ? '' : hit;
  let out = '';
  try {
    const files = DriveApp.getFolderById(folderId).getFiles();
    while (files.hasNext() && !out) {
      const f = files.next();
      const t = f.getThumbnail();
      if (t) out = 'data:' + (t.getContentType() || 'image/png') + ';base64,' + Utilities.base64Encode(t.getBytes());
    }
  } catch (_) {}
  try { cache.put('th_' + folderId, out && out.length < 95000 ? out : '-', 21600); } catch (_) {}
  return out.length < 95000 ? out : '';
}

 *
 *  הפרטים הסודיים לא נמצאים בקוד. מכניסים אותם פעם אחת ב:
 *  Project Settings (גלגל השיניים) > Script Properties

 *    SB_DOC_TYPE      (רשות) ברירת מחדל InvoiceReceipt = חשבונית מס קבלה
 *  בדיקה: להריץ את sbTest מהעורך ולראות את התוצאה ב-Execution log.

const SB_URLS = { test: 'https://test.smartbee.co.il/api/v1', prod: 'https://smartbee.co.il/api/v1' };

function sbProps_() { return PropertiesService.getScriptProperties(); }
function sbConfigured_() {
  const p = sbProps_();
  return !!(p.getProperty('SB_CLIENT_ID') && p.getProperty('SB_PASSWORD') && p.getProperty('SB_USER_TOKEN'));
}
function sbEnv_() { return sbProps_().getProperty('SB_ENV') === 'prod' ? 'prod' : 'test'; }
function sbBase_() { return SB_URLS[sbEnv_()]; }

/** טוקן גישה (בתוקף 7 ימים). נשמר ב-Script Properties ומתחדש לבד */
function sbToken_(force) {
  const p = sbProps_();
  const env = sbEnv_();
  const cached = p.getProperty('SB_TOKEN_' + env);
  const exp = Number(p.getProperty('SB_TOKEN_EXP_' + env) || 0);
  if (!force && cached && Date.now() < exp - 6 * 3600e3) return cached;
  const res = UrlFetchApp.fetch(sbBase_() + '/login/authenticate', {
    method: 'post', contentType: 'application/json', muteHttpExceptions: true,
    payload: JSON.stringify({ clientId: p.getProperty('SB_CLIENT_ID'), password: p.getProperty('SB_PASSWORD') }),
  });
  const code = res.getResponseCode();
  let body = {};
  try { body = JSON.parse(res.getContentText()); } catch (_) {}
  if (code !== 200 || !body.token) throw new Error('סמארטבי: ההתחברות נכשלה (' + code + '). לבדוק את SB_CLIENT_ID ו-SB_PASSWORD');
  const expMs = body.expirationUtcDate ? new Date(body.expirationUtcDate).getTime() : Date.now() + 6 * 864e5;
  p.setProperty('SB_TOKEN_' + env, body.token);
  p.setProperty('SB_TOKEN_EXP_' + env, String(expMs || Date.now() + 6 * 864e5));
  return body.token;
}

function sbFetch_(method, path, payload) {
  for (let attempt = 0; attempt < 3; attempt++) {
    const opt = { method, muteHttpExceptions: true, headers: { Authorization: 'Bearer ' + sbToken_(attempt === 1) } };
    if (payload) { opt.contentType = 'application/json'; opt.payload = JSON.stringify(payload); }
    const res = UrlFetchApp.fetch(sbBase_() + path, opt);
    const code = res.getResponseCode();
    let body = null;
    try { body = JSON.parse(res.getContentText()); } catch (_) { body = { raw: res.getContentText().slice(0, 300) }; }
    if (code === 401 && attempt === 0) continue;                       // טוקן פג: מתחברים מחדש פעם אחת
    if (code === 429 || code === 503) { Utilities.sleep(2000 * (attempt + 1)); continue; }
    if (code === 400) throw new Error('סמארטבי: נתונים לא תקינים: ' + sbErrText_(body));
    if (code >= 300) throw new Error('סמארטבי: שגיאה ' + code + ' ' + sbErrText_(body));
    return body;
  }
  throw new Error('סמארטבי לא זמינה כרגע. לנסות שוב בעוד כמה דקות');
}

function sbErrText_(b) {
  if (!b) return '';
  const v = b.validationErrors || b.errors || (b.result && b.result.validationErrors);
  if (v) return JSON.stringify(v).slice(0, 400);
  return String(b.title || b.message || b.raw || JSON.stringify(b)).slice(0, 400);
}

const SB_METHOD = { 'ביט': 'bit', 'העברה בנקאית': 'bank', 'מזומן': 'cash', 'אשראי': 'card' };

/**
 * מפיק חשבונית מס קבלה ומחכה לתוצאה.
 * p: { msgId, refId, customer:{...}, description, amount (לפני מע״מ), method ('ביט' / 'העברה בנקאית' / 'מזומן'), sendEmail }
 * מחזיר { index, link, copyLink, documentId, docType }
 */
function sbCreateInvoiceReceipt_(p) {
  const docType = sbProps_().getProperty('SB_DOC_TYPE') || 'InvoiceReceipt';
  const net = Math.round(Number(p.amount) * 100) / 100;
  if (!(net > 0)) throw new Error('סכום לא תקין');
  const gross = Math.round(net * (1 + CONFIG.VAT) * 100) / 100;
  const total = docType === 'Receipt' ? net : gross;           // עוסק פטור: קבלה בלי מע״מ

  const m = SB_METHOD[p.method] || 'other';
  const receiptDetails = {};
  if (m === 'cash') receiptDetails.cashItems = [{ sum: total }];
  else if (m === 'bit') receiptDetails.otherItems = [{ description: 'ביט', sum: total }];
  else if (m === 'bank') receiptDetails.otherItems = [{ description: 'העברה בנקאית', sum: total }];
  else receiptDetails.otherItems = [{ description: p.method || 'תשלום', sum: total }];

  const body = {
    providerUserToken: sbProps_().getProperty('SB_USER_TOKEN'),
    providerMsgId: p.msgId,
    providerMsgReferenceId: p.refId,
    docType,
    customer: p.customer,
    receiptDetails,
    creationMetadata: { language: 'HEB', sendOriginalToCustomer: p.sendEmail !== false },
    extraCommentsForEmail: 'תודה שאתם מפרסמים על מסך בית יעקב!',
  };
  if (docType !== 'Receipt') {
    body.documentItems = {
      paymentItems: [{ description: p.description, quantity: 1, pricePerUnit: net, vatOption: 'NotInclude', providerItemId: 'huri-screen-month' }],
      roundTotalSum: false,
    };
  }

  const created = sbFetch_('post', '/documents/create', body);
  const code = Number(created && created.resultCodeId);
  if (code !== 101 && code !== 95) throw new Error('סמארטבי לא קיבלה את הבקשה: ' + sbErrText_(created));
  const msgApiId = created.result;                                   // 95 = כבר נשלח בעבר; מחזיר את אותו מסמך

  for (let i = 0; i < 20; i++) {
    Utilities.sleep(i < 3 ? 1200 : 2500);
    const st = sbFetch_('get', '/documents/' + encodeURIComponent(msgApiId));
    const c = Number(st && st.resultCodeId);
    if (c === 1) continue;
    if (c === 102) {
      const r = st.result || {};
      return { index: r.index, link: r.linkToOriginal, copyLink: r.linkToCopy, documentId: r.documentId, docType: r.docType };
    }
    if (c === 103) throw new Error('סמארטבי יצרה טיוטה ולא מסמך סופי. לבדוק בממשק של סמארטבי');
    if (c === 96) throw new Error('סמארטבי: שגיאת נתונים: ' + sbErrText_(st));
    if (c === 94) throw new Error('סמארטבי: המשתמש לא מורשה. לבדוק את SB_USER_TOKEN');
    if (c === 104) throw new Error('סמארטבי: הפקת מסמכים חסומה בהגדרות המשתמש');
    if (c === 99 && docType === 'InvoiceReceipt') throw new Error('סמארטבי: סוג המסמך לא זמין לסוג העסק. אם אתה עוסק פטור, להגדיר SB_DOC_TYPE = Receipt');
    throw new Error('סמארטבי: שגיאה ' + c + ' ' + sbErrText_(st));
  }
  throw new Error('סמארטבי עדיין מעבדת את המסמך. לנסות "הפקת קבלה" שוב בעוד דקה (לא תיווצר כפולה)');
}

function sbCustomerFromRow_(get) {
  const c = { name: String(get('שם העסק')).slice(0, 100), providerCustomerId: 'HURI-' + get('מזהה') };

  const dealer = String(get('ח״פ / עוסק') || '').replace(/\D/g, ''); if (dealer) c.dealerNumber = dealer;

/**
 * רישום תשלום (העברה / ביט / מזומן) + הפקת חשבונית מס קבלה.
 * amount: סכום לפני מע״מ (ברירת מחדל: המחיר החודשי של הלקוח). activate: להפוך ממתין לפעיל.
 */
function adminRecordPayment(token, id, method, amount, issue, activate) {
  assertOwner_(token);
  const sh = getSheet_();
  const r = findRowById_(sh, id);
  if (!r) throw new Error('המפרסם לא נמצא');
  const row = sh.getRange(r, 1, 1, COLS.length).getValues()[0];
  const get = n => row[C[n] - 1];
  method = method || get('אמצעי תשלום') || PAY.BANK;
  const price = Number(get('מחיר חודשי')) || CONFIG.MONTHLY_PRICE;
  amount = Number(amount) > 0 ? Math.round(Number(amount) * 100) / 100 : price;

  if (activate) {
    sh.getRange(r, C['סטטוס']).setValue(STATUS.ACTIVE);
    if (!(get('תחילת פרסום') instanceof Date)) sh.getRange(r, C['תחילת פרסום']).setValue(startOfDay_(new Date()));
  }
  const start = sh.getRange(r, C['תחילת פרסום']).getValue();
  const cur = get('שולם עד');
  const from = cur instanceof Date ? cur : (start instanceof Date ? start : startOfDay_(new Date()));
  const to = addMonths_(from, 1);
  sh.getRange(r, C['שולם עד']).setValue(to);

  const fmt = d => Utilities.formatDate(d, CONFIG.TIMEZONE, 'dd/MM/yyyy');
  const period = Utilities.formatDate(from, CONFIG.TIMEZONE, 'yyyyMMdd');
  const pay = { period, from: fmt(from), to: fmt(new Date(to.getTime() - 864e5)), amount, method, date: fmt(new Date()) };
  sh.getRange(r, C['תשלום אחרון']).setValue(JSON.stringify(pay));
  sh.getRange(r, C['קבלה אחרונה']).setValue('');
  sh.getRange(r, C['קישור קבלה']).setValue('');
  noteAppend_(sh, r, 'התקבל תשלום ' + amount + ' ₪ + מע״מ ב' + method + ' (' + pay.from + '–' + pay.to + ')');
  SpreadsheetApp.flush();

  let receiptError = '';
  if (issue && method !== PAY.CARD) {
    if (!sbConfigured_()) receiptError = 'סמארטבי עוד לא מחוברת, הקבלה לא הופקה';
    else {
      try { sbIssueForRow_(sh, r); } catch (e) { receiptError = String(e && e.message || e); }
    }
  }
  const data = getAdminData(token);
  data.notice = receiptError ? { type: 'error', text: 'התשלום נרשם, אבל: ' + receiptError } : null;
  return data;
}
